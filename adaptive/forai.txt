# 2. Vision & Math Module: Uncertainty Calculation (Value Function)
# ==============================================================================
class UncertaintyEngine:
    def __init__(self, grid_size=(16, 16), image_size=(224, 224)):
        self.gh, self.gw = grid_size
        self.H, self.W = image_size
        self.num_patches = self.gh * self.gw
    
    def boxes_to_token_mask(self, boxes_xyxy_norm: torch.Tensor) -> torch.Tensor:
        """
        将多个归一化的Bounding Box转换为合并的Patch Mask
        Output: [num_patches], 1.0 inside any box, 0.0 outside all boxes
        """
        mask = torch.zeros(self.num_patches)
        patch_h = 1.0 / self.gh
        patch_w = 1.0 / self.gw
        
        if boxes_xyxy_norm is None or len(boxes_xyxy_norm) == 0:
            return mask

        # 遍历所有box
        for i in range(boxes_xyxy_norm.shape[0]):
            x1, y1, x2, y2 = boxes_xyxy_norm[i]
            
            # 确保值在有效范围内
            x1 = max(0.0, min(1.0, x1))
            y1 = max(0.0, min(1.0, y1))
            x2 = max(0.0, min(1.0, x2))
            y2 = max(0.0, min(1.0, y2))
            
            # 遍历所有patch计算是否在box内
            for j in range(self.num_patches):
                row = j // self.gw
                col = j % self.gw
                
                # Patch center
                px = (col + 0.5) * patch_w
                py = (row + 0.5) * patch_h
                
                if x1 <= px <= x2 and y1 <= py <= y2:
                    mask[j] = 1.0
        
        # 不需要归一化，按需归一化
        return mask

    def calculate_kl_divergence(self, attn_map: torch.Tensor, target_mask: torch.Tensor) -> float:
        """
        计算 VLA Attention 分布与 Grounding DINO 目标分布的 KL 散度
        Metric: KL(P || Q)
        P: VLA Attention (Model Prediction)
        Q: Ideal Distribution (Grounding DINO Mask)
        """
        # 1. 构造 Q (Ideal Distribution)
        # 加上 epsilon 防止除零，并归一化
        epsilon = 1e-9
        Q = target_mask + epsilon
        Q = Q / Q.sum()

        # 2. 处理 P (Attention Map)
        # attn_map shape 应该是 [num_patches]
        P = attn_map + epsilon
        P = P / P.sum()

        # 3. 计算 KL = sum(P * log(P / Q))
        # 如果 P 很确信地指向了 Q 之外的地方，KL 会很大
        kl_div = F.kl_div(P.log(), Q, reduction='mean').item() #batchmean是每个样本的特征和，mean是每个特征
        
        return kl_div

    def compute_uncertainty(self, 
                          vla_attn_layer: torch.Tensor, 
                          dino_boxes: torch.Tensor) -> Dict[str, float]:
        """
        Input:
            vla_attn_layer: [num_heads, seq_len] (Raw attention from VLA)
            dino_boxes: [N, 4] (Normalized xyxy from Grounding DINO)
        """
        # 1. 提取 Action Token 对 Image Tokens 的 Attention
        # 这里的索引需要根据 PI0.5 的具体 tokenizer 和 embedding 结构调整
        img_start, img_end = 0, self.num_patches
        action_token_idx = -1         
        # 平均所有 head 的注意力 (或者选取特定的 Expert Head)
        attn_avg = vla_attn_layer.mean(dim=0, keepdim=False)[img_start:img_end]
        attn_prob = attn_avg / attn_avg.sum() if attn_avg.sum() > 0 else attn_avg
        
        # 2. 生成目标掩码
        target_mask = self.boxes_to_token_mask(dino_boxes)
        
        masked = attn_prob * target_mask
        mib = masked.sum().item()
        
        max_val_global = attn_prob.max().item()
        max_val_in_target = masked.max().item()
        peak_ratio = max_val_in_target / (max_val_global + 1e-9)
        
        # 如果 DINO 没检测到东西，不确定性直接拉满
        if target_mask.sum() == 0:
            return {"kl": 10.0, "coverage": 0.0, "peak_ratio": 0.0, "uncertainty": 0.0}
        target_mask = target_mask / target_mask.sum() 
        
        uniform_mask = torch.ones(self.num_patches) / self.num_patches

        # 3. 计算指标
        kl = self.calculate_kl_divergence(attn_avg, target_mask)
        
        # 小于0更好
        uncertainty = self.calculate_kl_divergence(attn_avg, uniform_mask) - kl
        
        # 辅助指标：Attention Overlap (有多少注意力落在了框内)
        overlap = (attn_avg * target_mask).sum().item() / attn_avg.sum().item()
        
        return {"kl": kl, "coverage": mib, "peak_ratio": peak_ratio, "uncertainty": uncertainty}

# 3. Strategy Module: Adaptive Horizon Controller
# ==============================================================================
class AdaptiveActionChunker:
    def __init__(self, 
                 min_horizon=5, 
                 max_horizon=30, 
                 uncertainty_threshold=0.01,
                 coverage_threshold=0.1,
                 continuity_threshold=0.2,
                 action_history_window=5):
        
        self.uncertainty_engine = UncertaintyEngine()
        
        self.min_h = min_horizon
        self.max_h = max_horizon
        self.threshold = uncertainty_threshold
        self.coverage_threshold = coverage_threshold
        self.action_history_window = action_history_window
        self.continuity_threshold = continuity_threshold 
        # 状态记录
        self.current_horizon = (self.max_h + self.min_h) // 2
        self.history = []
        # 动作历史记录，用于计算轨迹连续性
        self.action_history = []
        self.prev_continuity = None
        # 打断标志，用于指示是否需要立即打断当前chunk
        self.needs_interruption = False
        
        # Grounding DINO model initialization
        self.config_path = '/pub/data/wangzb/GroundingDINO/groundingdino/config/GroundingDINO_SwinB_cfg.py'
        self.checkpoint_path = '/pub/data/wangzb/GroundingDINO/weights/20260314_pool_8~2_best.pth'
        # self.checkpoint_path = '/pub/data/wangzb/GroundingDINO/weights/groundingdino_swinb_cogcoor.pth'
        self.device = 'cuda' if torch.cuda.is_available() else 'cpu'
        
        # Load the model
        self.model = load_model(self.config_path, self.checkpoint_path, device=self.device)
        self.cached_boxes = None
        self.query_box_times = 0
        self.chunk_levels = 5
        
    def reset(self):
        """重置状态"""
        self.current_horizon = (self.max_h + self.min_h) // 2
        self.history = []
        self.action_history = []
        self.prev_continuity = None
        self.needs_interruption = False

    def get_dino_boxes_all(self, image, text_prompt):
        # if self.cached_boxes is not None and self.query_box_times % 10 == 0:
            # return self.cached_boxes
        self.query_box_times += 1
        """
        调用 Grounding DINO 模型进行目标检测（保留所有结果）
        Input:
            image: 图像数据 (numpy array 或路径)
            text_prompt: 文本提示，可以是字符串或对象列表
        Output:
            tuple: (boxes_cxcywh, boxes_xyxy, logits, phrases, image_source)
                boxes_cxcywh: torch.Tensor 归一化的边界框坐标 [N, 4] (cxcywh)
                boxes_xyxy: torch.Tensor 原始边界框坐标 [N, 4] (xyxy)
                logits: torch.Tensor 置信度 [N]
                phrases: list 识别的短语 [N]
                image_source: numpy array 原始图像
        """
        # 处理文本提示
        if isinstance(text_prompt, list):
            # 如果是对象列表，将其转换为逗号分隔的字符串
            prompt = " . ".join(text_prompt)
        else:
            prompt = text_prompt
        
        # 处理图像输入
        if isinstance(image, np.ndarray):
            # 如果是numpy数组，保存为临时文件
            import tempfile
            import os
            with tempfile.NamedTemporaryFile(suffix='.png', delete=False) as tmp:
                cv2.imwrite(tmp.name, image)
                image_path = tmp.name
        else:
            # 假设是图像路径
            image_path = image
        
        try:
            # 加载图像
            image_source, image = load_image(image_path)
            
            # 设置预测参数
            box_threshold = 0.3
            text_threshold = 0.25
            
            # 调用模型进行预测
            boxes, logits, phrases = predict(
                model=self.model,
                image=image,
                caption=prompt,
                box_threshold=box_threshold,
                text_threshold=text_threshold,
                device=self.device
            )
            
            # 将边界框转换为归一化的 cxcywh 格式
            if len(boxes) > 0:
                # 模型返回的是 [cx, cy, w, h] 格式
                # 需要转换为 [x1, y1, x2, y2] 格式用于可视化
                h, w = image_source.shape[:2]
                
                # 确保boxes在有效范围内
                boxes = torch.clamp(boxes, 0.0, 1.0)
                
                # 确保宽度和高度为正值
                boxes[:, 2] = torch.clamp(boxes[:, 2], min=0.01)
                boxes[:, 3] = torch.clamp(boxes[:, 3], min=0.01)
                
                # 提取中心点、宽度和高度
                cx = boxes[:, 0]
                cy = boxes[:, 1]
                width = boxes[:, 2]
                height = boxes[:, 3]
                
                # 转换为 [x1, y1, x2, y2] 格式用于可视化
                x1 = cx - width / 2
                y1 = cy - height / 2
                x2 = cx + width / 2
                y2 = cy + height / 2
                
                boxes_xyxy = torch.stack([x1, y1, x2, y2], dim=1)
                
                boxes_cxcywh = boxes
            else:
                # 如果没有检测到目标，返回空张量
                boxes_cxcywh = torch.tensor([])
                boxes_xyxy = torch.tensor([])
                
        finally:
            # 清理临时文件
            if 'image_path' in locals() and isinstance(image, np.ndarray):
                import os
                if os.path.exists(image_path):
                    os.unlink(image_path)
        # self.cached_boxes = (boxes_cxcywh, boxes_xyxy, logits, phrases, image_source)
        return boxes_cxcywh, boxes_xyxy, logits, phrases, image_source

    def get_dino_boxes(self, image, text_prompt):
        """
        调用 Grounding DINO 模型进行目标检测（去重，保留每个phrase的最高概率结果）
        Input:
            image: 图像数据 (numpy array 或路径)
            text_prompt: 文本提示，可以是字符串或对象列表
        Output:
            tuple: (boxes_cxcywh, boxes_xyxy, logits, phrases, image_source)
                boxes_cxcywh: torch.Tensor 归一化的边界框坐标 [N, 4] (cxcywh)
                boxes_xyxy: torch.Tensor 原始边界框坐标 [N, 4] (xyxy)
                logits: torch.Tensor 置信度 [N]
                phrases: list 识别的短语 [N]
                image_source: numpy array 原始图像
        """
        # 调用保留所有结果的方法
        boxes_cxcywh_all, boxes_all, logits_all, phrases_all, image_source = self.get_dino_boxes_all(image, text_prompt)
        
        # 去重：对于相同phrase的结果，只保留概率最高的那个
        if len(boxes_all) > 0:
            # 创建字典存储每个phrase的最高概率结果
            phrase_dict = {}
            
            for i, phrase in enumerate(phrases_all):
                if phrase not in phrase_dict or logits_all[i] > phrase_dict[phrase]['logit']:
                    phrase_dict[phrase] = {
                        'box': boxes_all[i],
                        'logit': logits_all[i],
                        'phrase': phrase,
                        'box_cxcywh': boxes_cxcywh_all[i]
                    }
            
            # 重新构建结果列表
            unique_boxes = []
            unique_logits = []
            unique_phrases = []
            unique_boxes_cxcywh = []
            
            for item in phrase_dict.values():
                unique_boxes.append(item['box'])
                unique_logits.append(item['logit'])
                unique_phrases.append(item['phrase'])
                unique_boxes_cxcywh.append(item['box_cxcywh'])
            
            # 转换为张量
            boxes = torch.stack(unique_boxes) if unique_boxes else torch.tensor([])
            logits = torch.tensor(unique_logits) if unique_logits else torch.tensor([])
            phrases = unique_phrases
            boxes_cxcywh = torch.stack(unique_boxes_cxcywh) if unique_boxes_cxcywh else torch.tensor([])
            
            # 计算boxes_xyxy
            if len(boxes) > 0:
                h, w = image_source.shape[:2]
                boxes_xyxy = boxes * torch.Tensor([w, h, w, h])
            else:
                boxes_xyxy = torch.tensor([])
        else:
            # 如果没有检测到目标，返回空张量
            boxes_cxcywh = torch.tensor([])
            boxes = torch.tensor([])
            logits = torch.tensor([])
            phrases = []
            boxes_xyxy = torch.tensor([])
        
        return boxes_cxcywh, boxes, logits, phrases, image_source

    def enqueue_history(self, history_action):
        """将历史记录添加到队列中"""
        if history_action is not None:
            self.action_history.append(history_action)
    
    
    def compute_continuity(self):
        continuity_factor = "High"
        if len(self.action_history) > self.action_history_window:
            self.action_history = self.action_history[-self.action_history_window:]
        continuity = 0.0
        # 计算轨迹连续性：时间窗口内相邻动作的绝对值差之和
        if len(self.action_history) == self.action_history_window:
            gripper_changed = False
            for i in range(1, len(self.action_history)):
                prev_action = np.array(self.action_history[i-1][:-1])
                curr_action = np.array(self.action_history[i][:-1])
                continuity += np.linalg.norm(curr_action - prev_action)
            if self.action_history[-1][-1] - self.action_history[-2][-1] > 1.5: #不abs了吧就用正的减去负的
                gripper_changed = True #闭合
            continuity /= (len(self.action_history) - 1)
            
            # 基于连续性调整chunk size
            # 连续性高（动作变化小）可以增大chunk，连续性低（动作变化大）需要减小chunk
            if continuity > self.continuity_threshold or (self.prev_continuity is not None and continuity - self.prev_continuity < -0.1): 
                continuity_factor = f"Low continuity: {continuity:.4f}, gripper_changed: {gripper_changed}, prev_continuity: {self.prev_continuity}"
            else:
                continuity_factor = "High trajectory continuity"
            logging.info(f"continuity: {continuity:.4f}, gripper_changed: {gripper_changed}, prev_continuity: {self.prev_continuity}")

            self.prev_continuity = continuity
        else:
            gripper_changed = False
            if len(self.action_history) > 1:
                if self.action_history[-1][-1] - self.action_history[-2][-1] > 1.5: #不abs了吧就用正的减去负的
                    gripper_changed = True #闭合
            logging.info(f"gripper_changed: {gripper_changed}")
            self.prev_continuity = None #感觉这个很关键
        return continuity_factor
    
    def set_interruption(self):
        self.needs_interruption = True
        self.action_history = []
    
    def ValueFunc(self, image, instruction, vla_attention_map, use_uncertainty=True, use_coverage=True):
        """
        核心函数：根据图像、指令和 VLA 内部状态计算当前的"策略价值"
        并返回推荐的 horizon。
        Input:
            image: 图像数据 (numpy array 或路径)
            instruction: 任务指令
            vla_attention_map: VLA 注意力图 [num_layer, num_heads, seq_len] / list of list of tensors
            use_uncertainty: 是否使用不确定性计算策略
            use_coverage: 是否使用覆盖率计算策略
        Output:
            tuple: (horizon, info_dict)
                horizon: 推荐的 horizon 值
                info_dict: 包含计算信息的字典
        """
        # 处理vla_attention_map是list of list of tensors的情况
        if isinstance(vla_attention_map, list) and len(vla_attention_map) > 0:
            # 检查是否是list of list of tensors
            if all(isinstance(layer, list) for layer in vla_attention_map) and all(isinstance(item, torch.Tensor) for layer in vla_attention_map for item in layer):
                # 转换为 [num_layer, num_heads, seq_len] 的张量
                vla_attention_map = torch.stack([torch.stack(layer) for layer in vla_attention_map])
            elif all(isinstance(item, torch.Tensor) for item in vla_attention_map):
                # 兼容旧格式：list of tensors
                vla_attention_map = torch.stack(vla_attention_map)

        self.needs_interruption = False        
        
        # 1. 解析指令，获取 DINO 的 Prompt
        parsed_objs = extract_objects(instruction)
        if not parsed_objs:
            # 如果解析失败，保守策略
            return self.min_h, {"reason": "parse_fail"}

        # 2. 获取 Ground Truth (DINO)
        _, boxes_xyxy, _, phrases, _ = self.get_dino_boxes(image, parsed_objs)  
              
        # if len(phrases) != len(parsed_objs): #感觉不能这样判断
        #     logging.warning(f"DINO 识别的短语数量与解析的对象数量不匹配！parsed_objs: {parsed_objs}, phrases: {phrases}")
        #     return self.current_horizon, {"reason": "dino_parse_mismatch", "parsed_objs": parsed_objs, "dino_phrases": phrases}
        
        boxes = boxes_xyxy
        if len(boxes) > 0:
            boxes = torch.clamp(boxes, 0.0, 1.0)
        
        logging.info(f"parsed_objs: {parsed_objs}, boxes: {boxes}, phrases: {phrases}")

        # 3. 计算指标
        info_dict = {
            "target_object": parsed_objs,
            "dino_boxes": boxes.tolist()
        }
        
        confidence = 1.0
        reasons = []
        
        kl_scores = []
        overlaps = []
        peak_scores = []
        uncertainty_scores = []
        peak_high_count = 0
        peak_low_count = 0
        peak_thresh_high = 0.8
        peak_thresh_low = 0.3
        
        continuity_factor = self.compute_continuity()
        
        # 遍历每一层
        for layer_idx in range(vla_attention_map.shape[0]):
            layer_attn = vla_attention_map[layer_idx]
            metrics = self.uncertainty_engine.compute_uncertainty(layer_attn, boxes)
            logging.info(f"layer_idx: {layer_idx}, metrics: {metrics}")
            kl_scores.append(metrics['kl'])
            overlaps.append(metrics.get("coverage", 0.0))
            peak_scores.append(metrics['peak_ratio'])
            uncertainty_scores.append(metrics['uncertainty'])
            if peak_scores[-1] > peak_thresh_high:
                peak_high_count += 1
            elif peak_scores[-1] < peak_thresh_low:
                peak_low_count += 1
            # logging.info(metrics)
            
        # if peak_high_count >= 2:
        #     confidence = 5
        # elif continuity_factor.startswith("High"):
        #     confidence = 3
        
        # if peak_low_count >= 2 and continuity_factor.startswith("Low"):
        #     confidence = 1
        num_layers = len(peak_scores)
        peak_high_ratio = peak_high_count / max(num_layers, 1)
        peak_low_ratio = peak_low_count / max(num_layers, 1)
        avg_coverage = float(np.mean(overlaps)) if overlaps else 0.0
        best_kl = float(np.min(kl_scores)) if kl_scores else 10.0
        avg_uncertainty = float(np.mean(uncertainty_scores)) if uncertainty_scores else -1.0

        is_high_cont = continuity_factor.startswith("High")
        is_low_cont = continuity_factor.startswith("Low")

        confidence = 2  # default -> chunk 15
        # 不想用kl来判断

        # if is_low_cont and (peak_low_ratio >= 0.75 or avg_coverage < 0.05 or best_kl > 0.02):
        #     confidence = 0
        #     self.needs_interruption = True
        # elif is_low_cont or peak_low_ratio >= 0.5 or avg_coverage < 0.08 or best_kl > 0.018:
        #     confidence = 1
        # elif peak_high_ratio >= 0.5 and avg_coverage >= 0.20 and best_kl <= 0.008 and is_high_cont:
        #     confidence = 5
        # elif peak_high_ratio >= 0.5 and avg_coverage >= 0.14 and best_kl <= 0.012:
        #     confidence = 4
        # elif peak_high_ratio >= 0.25 and avg_coverage >= 0.10 and best_kl <= 0.016:
        #     confidence = 3
        # elif is_high_cont:
        #     confidence = 3
        # else:
        #     confidence = 2
        if is_low_cont and (peak_low_ratio >= 0.75 or avg_coverage < 0.05):
            confidence = 0
            # self.needs_interruption = True
        elif is_low_cont or peak_low_ratio >= 0.5 or avg_coverage < 0.08:
            confidence = 1
        elif peak_high_ratio >= 0.5 and avg_coverage >= 0.20 and is_high_cont:
            confidence = 5
        elif peak_high_ratio >= 0.5 and avg_coverage >= 0.14:
            confidence = 4
        elif peak_high_ratio >= 0.25 and avg_coverage >= 0.10 :
            confidence = 3
        elif is_high_cont:
            confidence = 3
        else:
            confidence = 2

        
        logging.info(f"confidence: {confidence}, continuity_factor: {continuity_factor}")
                
        prev_h = self.current_horizon        
        self.current_horizon = max(self.min_h, self.get_chunk_size(confidence))
        
        return self.current_horizon, info_dict
    
    def get_chunk_size(self, confidence): # 0, 1, 2, 3, 4, 5
        return int(self.min_h + (self.max_h - self.min_h) / self.chunk_levels * confidence)


    def adaptive_chunk_heuristic(self, image, instruction, robot_state=None, object_state=None):
        """
        基于启发式逻辑的自适应chunk size调整函数
        不依赖VLA attention map，可在后台持续运行
        Input:
            image: 图像数据 (numpy array 或路径)
            instruction: 任务指令
            robot_state: 机器人状态 (可选)，包含位置、速度等信息
            object_state: 对象状态 (可选)，包含位置、速度等信息
            current_action: 当前动作 (可选)，用于计算轨迹连续性
        Output:
            tuple: (horizon, info_dict)
                horizon: 推荐的 horizon 值
                info_dict: 包含计算信息的字典
        """
        # 1. 解析指令，获取 DINO 的 Prompt
        parsed_objs = extract_objects(instruction)
        
        if not parsed_objs:
            # 如果解析失败，保守策略
            return self.min_h, {"reason": "parse_fail"}

        # 2. 获取 Ground Truth (DINO)
        boxes_cxcywh, _, _, _, _ = self.get_dino_boxes(image, parsed_objs)
        boxes = boxes_cxcywh
        
        # 3. 初始化信息字典
        info_dict = {
            "target_object": parsed_objs,
            "dino_boxes": boxes.tolist()
        }
        
        factors = []
        # 4.1 基于目标检测的启发式
        if len(boxes) > 0:
            # 目标数量因子：目标越多，需要更频繁的决策
            num_objects = len(boxes)
            info_dict["num_objects"] = num_objects
            if num_objects > 3:
                factors.append("Multiple objects detected")
        
        
        # 4.2 基于机器人状态的启发式
        if robot_state:
            # 末端执行器与目标的距离：距离越近，需要更频繁的决策
            if "ee_position" in robot_state and len(boxes) > 0:
                # 计算末端执行器与最近目标的距离
                ee_pos = np.array(robot_state["ee_position"])
                min_distance = float('inf')
                for box in boxes:
                    # 假设box是 [cx, cy, w, h] 归一化坐标
                    box_center = np.array([box[0], box[1]])
                    distance = np.linalg.norm(ee_pos[:2] - box_center)
                    if distance < min_distance:
                        min_distance = distance
                
                info_dict["min_object_distance"] = min_distance
                if min_distance < 0.1:  # 距离阈值，可调整
                    factors.append("Close to object (grasping phase)")
        
        
        # if factors:
        #     # 根据因子数量调整降低幅度
        #     factor_count = len(factors)
        #     logging.info(f"factor_count: {factor_count} factors: {factors}")
        #     reduction_factor = max(0.4, 1.0 - factor_count * 0.3)  # 最多降低60%
        #     self.current_horizon = max(self.min_h, int(prev_h * reduction_factor))
        #     reason = "; ".join(factors)
        # 5. 基于启发式因子调整horizon
        
        self.needs_interruption = False
        prev_h = self.current_horizon        
        continuity_factor = self.compute_continuity()
            
        if continuity_factor:
            if continuity_factor.startswith("High"):
                self.current_horizon = min(self.max_h, prev_h + 5)
                reason = continuity_factor
            else:
                self.set_interruption()
                logging.info("confidence: 2")
                self.current_horizon = max(self.min_h, self.get_chunk_size(2))
        
        factors.append(continuity_factor)
        info_dict["factors"] = factors
        
        return self.current_horizon, info_dict

class UncertaintyEngine:
    def __init__(self, grid_size=(16, 16), image_size=(224, 224)):
        self.gh, self.gw = grid_size
        self.H, self.W = image_size
        self.num_patches = self.gh * self.gw
    
    def boxes_to_token_mask(self, boxes_xyxy_norm: torch.Tensor) -> torch.Tensor:
        """
        将多个归一化的Bounding Box转换为合并的Patch Mask
        Output: [num_patches], 1.0 inside any box, 0.0 outside all boxes
        """
        mask = torch.zeros(self.num_patches)
        patch_h = 1.0 / self.gh
        patch_w = 1.0 / self.gw
        
        if boxes_xyxy_norm is None or len(boxes_xyxy_norm) == 0:
            return mask

        # 遍历所有box
        for i in range(boxes_xyxy_norm.shape[0]):
            x1, y1, x2, y2 = boxes_xyxy_norm[i]
            
            # 确保值在有效范围内
            x1 = max(0.0, min(1.0, x1))
            y1 = max(0.0, min(1.0, y1))
            x2 = max(0.0, min(1.0, x2))
            y2 = max(0.0, min(1.0, y2))
            
            # 遍历所有patch计算是否在box内
            for j in range(self.num_patches):
                row = j // self.gw
                col = j % self.gw
                
                # Patch center
                px = (col + 0.5) * patch_w
                py = (row + 0.5) * patch_h
                
                if x1 <= px <= x2 and y1 <= py <= y2:
                    mask[j] = 1.0
        
        # 不需要归一化，按需归一化
        return mask

    def calculate_kl_divergence(self, attn_map: torch.Tensor, target_mask: torch.Tensor) -> float:
        """
        计算 VLA Attention 分布与 Grounding DINO 目标分布的 KL 散度
        Metric: KL(P || Q)
        P: VLA Attention (Model Prediction)
        Q: Ideal Distribution (Grounding DINO Mask)
        """
        # 1. 构造 Q (Ideal Distribution)
        # 加上 epsilon 防止除零，并归一化
        epsilon = 1e-9
        Q = target_mask + epsilon
        Q = Q / Q.sum()

        # 2. 处理 P (Attention Map)
        # attn_map shape 应该是 [num_patches]
        P = attn_map + epsilon
        P = P / P.sum()

        # 3. 计算 KL = sum(P * log(P / Q))
        # 如果 P 很确信地指向了 Q 之外的地方，KL 会很大
        kl_div = F.kl_div(P.log(), Q, reduction='mean').item() #batchmean是每个样本的特征和，mean是每个特征
        
        return kl_div

    def compute_uncertainty(self, 
                          vla_attn_layer: torch.Tensor, 
                          dino_boxes: torch.Tensor) -> Dict[str, float]:
        """
        Input:
            vla_attn_layer: [num_heads, seq_len] (Raw attention from VLA)
            dino_boxes: [N, 4] (Normalized xyxy from Grounding DINO)
        """
        # 1. 提取 Action Token 对 Image Tokens 的 Attention
        # 这里的索引需要根据 PI0.5 的具体 tokenizer 和 embedding 结构调整
        img_start, img_end = 0, self.num_patches
        action_token_idx = -1         
        # 平均所有 head 的注意力 (或者选取特定的 Expert Head)
        attn_avg = vla_attn_layer.mean(dim=0, keepdim=False)[img_start:img_end]
        attn_prob = attn_avg / attn_avg.sum() if attn_avg.sum() > 0 else attn_avg
        
        # 2. 生成目标掩码
        target_mask = self.boxes_to_token_mask(dino_boxes)
        
        masked = attn_prob * target_mask
        mib = masked.sum().item()
        
        max_val_global = attn_prob.max().item()
        max_val_in_target = masked.max().item()
        peak_ratio = max_val_in_target / (max_val_global + 1e-9)
        
        # 如果 DINO 没检测到东西，不确定性直接拉满
        if target_mask.sum() == 0:
            return {"kl": 10.0, "entropy": -1.0, "overlap": 0.0}
        target_mask = target_mask / target_mask.sum() 
        
        uniform_mask = torch.ones(self.num_patches) / self.num_patches

        # 3. 计算指标
        kl = self.calculate_kl_divergence(attn_avg, target_mask)
        
        # 小于0更好
        uncertainty = self.calculate_kl_divergence(attn_avg, uniform_mask) - kl
        
        # 辅助指标：Attention Overlap (有多少注意力落在了框内)
        overlap = (attn_avg * target_mask).sum().item() / attn_avg.sum().item()
        
        return {"kl": kl, "coverage": mib, "peak_ratio": peak_ratio, "uncertainty": uncertainty}

class AdaptiveActionChunker:
    def __init__(self, 
                 min_horizon=5, 
                 max_horizon=30, 
                 uncertainty_threshold=0.01,
                 coverage_threshold=0.1,
                 continuity_threshold=0.2,
                 action_history_window=5):
        
        self.uncertainty_engine = UncertaintyEngine()
        
        self.min_h = min_horizon
        self.max_h = max_horizon
        self.threshold = uncertainty_threshold
        self.coverage_threshold = coverage_threshold
        self.action_history_window = action_history_window
        self.continuity_threshold = continuity_threshold 
        # 状态记录
        self.current_horizon = max_horizon // 2
        self.history = []
        # 动作历史记录，用于计算轨迹连续性
        self.action_history = []
        self.prev_continuity = None
        # 打断标志，用于指示是否需要立即打断当前chunk
        self.needs_interruption = False
        
        # Grounding DINO model initialization
        self.config_path = '/pub/data/wangzb/GroundingDINO/groundingdino/config/GroundingDINO_SwinB_cfg.py'

        self.device = 'cuda' if torch.cuda.is_available() else 'cpu'

    def reset(self):

    def get_dino_boxes_all(self, image, text_prompt):
        # if self.cached_boxes is not None and self.query_box_times % 10 == 0:
            # return self.cached_boxes
        self.query_box_times += 1
        """
        调用 Grounding DINO 模型进行目标检测（保留所有结果）
        Input:
            image: 图像数据 (numpy array 或路径)
            text_prompt: 文本提示，可以是字符串或对象列表
        Output:
            tuple: (boxes_cxcywh, boxes_xyxy, logits, phrases, image_source)
                boxes_cxcywh: torch.Tensor 归一化的边界框坐标 [N, 4] (cxcywh)
                boxes_xyxy: torch.Tensor 原始边界框坐标 [N, 4] (xyxy)
                logits: torch.Tensor 置信度 [N]
                phrases: list 识别的短语 [N]
                image_source: numpy array 原始图像
        """
        # 处理文本提示
        if isinstance(text_prompt, list):
            # 如果是对象列表，将其转换为逗号分隔的字符串
            prompt = " . ".join(text_prompt)
        else:
            prompt = text_prompt
        
        # 处理图像输入
        if isinstance(image, np.ndarray):
            # 如果是numpy数组，保存为临时文件
            import tempfile
            import os
            with tempfile.NamedTemporaryFile(suffix='.png', delete=False) as tmp:
                cv2.imwrite(tmp.name, image)
                image_path = tmp.name
        else:
            # 假设是图像路径
            image_path = image
        
        try:
            # 加载图像
            image_source, image = load_image(image_path)
            
            # 设置预测参数
            box_threshold = 0.3
            text_threshold = 0.25
            
            # 调用模型进行预测
            boxes, logits, phrases = predict(
                model=self.model,
                image=image,
                caption=prompt,
                box_threshold=box_threshold,
                text_threshold=text_threshold,
                device=self.device
            )
            
            # 将边界框转换为归一化的 cxcywh 格式
            if len(boxes) > 0:
                # 模型返回的是 [cx, cy, w, h] 格式
                # 需要转换为 [x1, y1, x2, y2] 格式用于可视化
                h, w = image_source.shape[:2]
                
                # 确保boxes在有效范围内
                boxes = torch.clamp(boxes, 0.0, 1.0)
                
                # 确保宽度和高度为正值
                boxes[:, 2] = torch.clamp(boxes[:, 2], min=0.01)
                boxes[:, 3] = torch.clamp(boxes[:, 3], min=0.01)
                
                # 提取中心点、宽度和高度
                cx = boxes[:, 0]
                cy = boxes[:, 1]
                width = boxes[:, 2]
                height = boxes[:, 3]
                
                # 转换为 [x1, y1, x2, y2] 格式用于可视化
                x1 = cx - width / 2
                y1 = cy - height / 2
                x2 = cx + width / 2
                y2 = cy + height / 2
                
                boxes_xyxy = torch.stack([x1, y1, x2, y2], dim=1)
                
                boxes_cxcywh = boxes
            else:
                # 如果没有检测到目标，返回空张量
                boxes_cxcywh = torch.tensor([])
                boxes_xyxy = torch.tensor([])
                
        finally:
            # 清理临时文件
            if 'image_path' in locals() and isinstance(image, np.ndarray):
                import os
                if os.path.exists(image_path):
                    os.unlink(image_path)
        # self.cached_boxes = (boxes_cxcywh, boxes_xyxy, logits, phrases, image_source)
        return boxes_cxcywh, boxes_xyxy, logits, phrases, image_source

    def get_dino_boxes(self, image, text_prompt):
        """
        调用 Grounding DINO 模型进行目标检测（去重，保留每个phrase的最高概率结果）
        Input:
            image: 图像数据 (numpy array 或路径)
            text_prompt: 文本提示，可以是字符串或对象列表
        Output:
            tuple: (boxes_cxcywh, boxes_xyxy, logits, phrases, image_source)
                boxes_cxcywh: torch.Tensor 归一化的边界框坐标 [N, 4] (cxcywh)
                boxes_xyxy: torch.Tensor 原始边界框坐标 [N, 4] (xyxy)
                logits: torch.Tensor 置信度 [N]
                phrases: list 识别的短语 [N]
                image_source: numpy array 原始图像
        """
        # 调用保留所有结果的方法
        boxes_cxcywh_all, boxes_all, logits_all, phrases_all, image_source = self.get_dino_boxes_all(image, text_prompt)
        
        # 去重：对于相同phrase的结果，只保留概率最高的那个
        if len(boxes_all) > 0:
            # 创建字典存储每个phrase的最高概率结果
            phrase_dict = {}
            
            for i, phrase in enumerate(phrases_all):
                if phrase not in phrase_dict or logits_all[i] > phrase_dict[phrase]['logit']:
                    phrase_dict[phrase] = {
                        'box': boxes_all[i],
                        'logit': logits_all[i],
                        'phrase': phrase,
                        'box_cxcywh': boxes_cxcywh_all[i]
                    }
            
            # 重新构建结果列表
            unique_boxes = []
            unique_logits = []
            unique_phrases = []
            unique_boxes_cxcywh = []
            
            for item in phrase_dict.values():
                unique_boxes.append(item['box'])
                unique_logits.append(item['logit'])
                unique_phrases.append(item['phrase'])
                unique_boxes_cxcywh.append(item['box_cxcywh'])
            
            # 转换为张量
            boxes = torch.stack(unique_boxes) if unique_boxes else torch.tensor([])
            logits = torch.tensor(unique_logits) if unique_logits else torch.tensor([])
            phrases = unique_phrases
            boxes_cxcywh = torch.stack(unique_boxes_cxcywh) if unique_boxes_cxcywh else torch.tensor([])
            
            # 计算boxes_xyxy
            if len(boxes) > 0:
                h, w = image_source.shape[:2]
                boxes_xyxy = boxes * torch.Tensor([w, h, w, h])
            else:
                boxes_xyxy = torch.tensor([])
        else:
            # 如果没有检测到目标，返回空张量
            boxes_cxcywh = torch.tensor([])
            boxes = torch.tensor([])
            logits = torch.tensor([])
            phrases = []
            boxes_xyxy = torch.tensor([])
        
        return boxes_cxcywh, boxes, logits, phrases, image_source

    def enqueue_history(self, history_action):
        """将历史记录添加到队列中"""
        if history_action is not None:
            self.action_history.append(history_action)
    
    
    def compute_continuity(self):
        continuity_factor = "High"
        if len(self.action_history) > self.action_history_window:
            self.action_history = self.action_history[-self.action_history_window:]
        continuity = 0.0
        # 计算轨迹连续性：时间窗口内相邻动作的绝对值差之和
        if len(self.action_history) == self.action_history_window:
            # gripper_changed1 = False
            # gripper_changed2 = False
            gripper_changed = False
            for i in range(1, len(self.action_history)):
                prev_action = np.array(self.action_history[i-1][:-1])
                curr_action = np.array(self.action_history[i][:-1])
                continuity += np.linalg.norm(curr_action - prev_action)
                # if self.action_history[i][-1] - self.action_history[i-1][-1] > 1.5: #不abs了吧就用正的减去负的
                #     gripper_changed1 = True #闭合
                # if self.action_history[i][-1] - self.action_history[i-1][-1] < -1.5:
                #     gripper_changed2 = True # 不行这样的话在历史中有>1.5的，大概率就有<-1.5的了，按顺序出现
            # 只触发一次，不多次了, 这个也会导致不少问题，继续改用预测的来做
            if self.action_history[-1][-1] - self.action_history[-2][-1] > 1.5: #不abs了吧就用正的减去负的
                gripper_changed = True #闭合
            continuity /= (len(self.action_history) - 1)
            
            # logging.info(f"trajectory_continuity: {continuity:.4f}")
            
            # 基于连续性调整chunk size
            # 连续性高（动作变化小）可以增大chunk，连续性低（动作变化大）需要减小chunk
            # continuity_threshold = 0.2  # 可调整的阈值
            if continuity > self.continuity_threshold or (self.prev_continuity is not None and continuity - self.prev_continuity < -0.1): 
                continuity_factor = f"Low continuity: {continuity:.4f}, gripper_changed: {gripper_changed}, prev_continuity: {self.prev_continuity}"
                self.needs_interruption = True
                self.action_history = []
            else:
                continuity_factor = "High trajectory continuity"
            # if self.prev_continuity is not None:
            #     logging.info(f"continuity: {continuity:.4f}, prev_continuity: {self.prev_continuity:.4f}")                
            # else:
            #     logging.info(f"continuity: {continuity:.4f}")
            logging.info(f"continuity: {continuity:.4f}, gripper_changed: {gripper_changed}, prev_continuity: {self.prev_continuity}")

            self.prev_continuity = continuity
        else:
            gripper_changed = False
            if len(self.action_history) > 1:
                if self.action_history[-1][-1] - self.action_history[-2][-1] > 1.5: #不abs了吧就用正的减去负的
                    gripper_changed = True #闭合
            logging.info(f"gripper_changed: {gripper_changed}")
            self.prev_continuity = None #感觉这个很关键
        return continuity_factor
        
    
    def ValueFunc(self, image, instruction, vla_attention_map, use_uncertainty=True, use_coverage=True):
        """
        核心函数：根据图像、指令和 VLA 内部状态计算当前的"策略价值"
        并返回推荐的 horizon。
        Input:
            image: 图像数据 (numpy array 或路径)
            instruction: 任务指令
            vla_attention_map: VLA 注意力图 [num_layer, num_heads, seq_len] / list of list of tensors
            use_uncertainty: 是否使用不确定性计算策略
            use_coverage: 是否使用覆盖率计算策略
        Output:
            tuple: (horizon, info_dict)
                horizon: 推荐的 horizon 值
                info_dict: 包含计算信息的字典
        """
        # 处理vla_attention_map是list of list of tensors的情况
        if isinstance(vla_attention_map, list) and len(vla_attention_map) > 0:
            # 检查是否是list of list of tensors
            if all(isinstance(layer, list) for layer in vla_attention_map) and all(isinstance(item, torch.Tensor) for layer in vla_attention_map for item in layer):
                # 转换为 [num_layer, num_heads, seq_len] 的张量
                vla_attention_map = torch.stack([torch.stack(layer) for layer in vla_attention_map])
            elif all(isinstance(item, torch.Tensor) for item in vla_attention_map):
                # 兼容旧格式：list of tensors
                vla_attention_map = torch.stack(vla_attention_map)

        self.needs_interruption = False        
        
        # 1. 解析指令，获取 DINO 的 Prompt
        parsed_objs = extract_objects(instruction)
        if not parsed_objs:
            # 如果解析失败，保守策略
            return self.min_h, {"reason": "parse_fail"}

        # 2. 获取 Ground Truth (DINO)
        _, boxes_xyxy, _, phrases, _ = self.get_dino_boxes(image, parsed_objs)        
        if len(phrases) != len(parsed_objs):
            logging.warning(f"DINO 识别的短语数量与解析的对象数量不匹配！parsed_objs: {parsed_objs}, phrases: {phrases}")
            return self.current_horizon, {"reason": "dino_parse_mismatch", "parsed_objs": parsed_objs, "dino_phrases": phrases}
        
        boxes = boxes_xyxy
        if len(boxes) > 0:
            boxes = torch.clamp(boxes, 0.0, 1.0)
        
        logging.info(f"parsed_objs: {parsed_objs}, boxes: {boxes}, phrases: {phrases}")

        # 3. 计算指标
        info_dict = {
            "target_object": parsed_objs,
            "dino_boxes": boxes.tolist()
        }
        
        confidence = 1.0
        reasons = []
        
        kl_scores = []
        overlaps = []
        peak_scores = []
        uncertainty_scores = []
        peak_high_count = 0
        peak_low_count = 0
        peak_thresh_high = 0.8
        peak_thresh_low = 0.4
        
        continuity_factor = self.compute_continuity()
        
        # 遍历每一层
        for layer_idx in range(vla_attention_map.shape[0]):
            layer_attn = vla_attention_map[layer_idx]
            metrics = self.uncertainty_engine.compute_uncertainty(layer_attn, boxes)
            logging.info(f"layer_idx: {layer_idx}, metrics: {metrics}")
            kl_scores.append(metrics['kl'])
            overlaps.append(metrics.get("coverage", 0.0))
            peak_scores.append(metrics['peak_ratio'])
            uncertainty_scores.append(metrics['uncertainty'])
            if peak_scores[-1] > peak_thresh_high:
                peak_high_count += 1
            elif peak_scores[-1] < peak_thresh_low:
                peak_low_count += 1
            # logging.info(metrics)
            
        if peak_high_count >= 2:
            confidence = 1.0
        elif continuity_factor.startswith("High"):
            confidence = 0.7
        
        if peak_low_count >= 2 and continuity_factor.startswith("Low"):
            confidence = 0.3
            self.needs_interruption = True
        
        logging.info(f"confidence: {confidence}, continuity_factor: {continuity_factor}")
                
        prev_h = self.current_horizon        
        self.current_horizon = max(self.min_h, int(prev_h * confidence))
        
        return self.current_horizon, info_dict
    
    def get_chunk_size(self, confidence):
        return int(self.min_h + (self.max_h - self.min_h) * confidence)


    def adaptive_chunk_heuristic(self, image, instruction, robot_state=None, object_state=None):
        """
        基于启发式逻辑的自适应chunk size调整函数
        不依赖VLA attention map，可在后台持续运行
        Input:
            image: 图像数据 (numpy array 或路径)
            instruction: 任务指令
            robot_state: 机器人状态 (可选)，包含位置、速度等信息
            object_state: 对象状态 (可选)，包含位置、速度等信息
            current_action: 当前动作 (可选)，用于计算轨迹连续性
        Output:
            tuple: (horizon, info_dict)
                horizon: 推荐的 horizon 值
                info_dict: 包含计算信息的字典
        """
        # 1. 解析指令，获取 DINO 的 Prompt
        parsed_objs = extract_objects(instruction)
        
        if not parsed_objs:
            # 如果解析失败，保守策略
            return self.min_h, {"reason": "parse_fail"}

        # 2. 获取 Ground Truth (DINO)
        boxes_cxcywh, _, _, _, _ = self.get_dino_boxes(image, parsed_objs)
        boxes = boxes_cxcywh
        
        # 3. 初始化信息字典
        info_dict = {
            "target_object": parsed_objs,
            "dino_boxes": boxes.tolist()
        }
        
        factors = []
        # 4.1 基于目标检测的启发式
        if len(boxes) > 0:
            # 目标数量因子：目标越多，需要更频繁的决策
            num_objects = len(boxes)
            info_dict["num_objects"] = num_objects
            if num_objects > 3:
                factors.append("Multiple objects detected")
        
        self.needs_interruption = False
        
        # 4.2 基于机器人状态的启发式
        if robot_state:
            # 末端执行器速度：速度越快，需要更频繁的决策
            
            # 末端执行器与目标的距离：距离越近，需要更频繁的决策
            if "ee_position" in robot_state and len(boxes) > 0:
                # 计算末端执行器与最近目标的距离
                ee_pos = np.array(robot_state["ee_position"])
                min_distance = float('inf')
                for box in boxes:
                    # 假设box是 [cx, cy, w, h] 归一化坐标
                    # 转换为实际坐标
                    box_center = np.array([box[0], box[1]])
                    distance = np.linalg.norm(ee_pos[:2] - box_center)
                    if distance < min_distance:
                        min_distance = distance
                
                info_dict["min_object_distance"] = min_distance
                if min_distance < 0.1:  # 距离阈值，可调整
                    factors.append("Close to object (grasping phase)")
        
        # 5. 基于启发式因子调整horizon
        prev_h = self.current_horizon
        
        continuity_factor = self.compute_continuity()
        
        if factors:
            # 根据因子数量调整降低幅度
            factor_count = len(factors)
            reduction_factor = max(0.4, 1.0 - factor_count * 0.3)  # 最多降低60%
            self.current_horizon = max(self.min_h, int(prev_h * reduction_factor))
            reason = "; ".join(factors)
        else:
            if continuity_factor:
                if continuity_factor.startswith("High"):
                    self.current_horizon = min(self.max_h, prev_h + 3)
                    reason = continuity_factor
                else:
                    self.current_horizon = max(self.min_h, int(prev_h * 0.5))
        
        factors.append(continuity_factor)
        info_dict["factors"] = factors
        
        return self.current_horizon, info_dict

class AdaptiveActionChunker(TrackerAdaptiveActionChunker):
    """
    Experimental score-gated tracker policy.

    Differences from the default tracker policy:
    - DINO is treated as a candidate during refresh and is not committed first.
    - When tracker and DINO disagree, prefer DINO only if its score is higher
      than the tracked score by `dino_score_margin`.
    """

    def __init__(self, *args, dino_score_margin: float = 0.1, **kwargs):
        super().__init__(*args, **kwargs)
        self.dino_score_margin = float(dino_score_margin)

    def _score_value(self, logits) -> float:
        if logits is None:
            return float("-inf")
        try:
            if hasattr(logits, "detach"):
                logits = logits.detach()
            if hasattr(logits, "cpu"):
                logits = logits.cpu()
            if hasattr(logits, "numpy"):
                logits = logits.numpy()
            if hasattr(logits, "tolist"):
                logits = logits.tolist()
            if isinstance(logits, (list, tuple)):
                values = []
                for item in logits:
                    try:
                        if hasattr(item, "item"):
                            item = item.item()
                        values.append(float(item))
                    except Exception:
                        continue
                return max(values) if values else float("-inf")
            if hasattr(logits, "item"):
                logits = logits.item()
            return float(logits)
        except Exception:
            return float("-inf")

    def _score_map(self, phrases, logits):
        score_by_phrase = {}
        if phrases is None or logits is None:
            return score_by_phrase
        for idx, phrase in enumerate(phrases):
            try:
                value = logits[idx]
            except Exception:
                continue
            try:
                if hasattr(value, "detach"):
                    value = value.detach()
                if hasattr(value, "cpu"):
                    value = value.cpu()
                if hasattr(value, "item"):
                    value = value.item()
                score_by_phrase[phrase] = float(value)
            except Exception:
                continue
        return score_by_phrase

    def _has_boxes(self, result) -> bool:
        try:
            boxes_xyxy = result[1]
            return boxes_xyxy is not None and len(boxes_xyxy) > 0
        except Exception:
            return False

    def _commit_result(self, image, result, prompt_key, used_dino: bool):
        boxes_cxcywh, boxes_xyxy, logits, phrases, _ = result
        cached_image = self._ensure_frame(image)
        self._set_cached_boxes(
            boxes_cxcywh,
            boxes_xyxy,
            logits,
            phrases,
            cached_image,
        )
        self._initialize_trackers(image, boxes_xyxy, phrases, prompt_key)
        self.last_query_used_dino = used_dino
        return self.cached_boxes

    def _raw_dino_candidate(self, image, text_prompt):
        return BaseAdaptiveActionChunker.get_dino_boxes(self, image, text_prompt)

    def _phrase_maps(self, result):
        boxes_xyxy = result[1]
        logits = result[2]
        phrases = result[3]
        boxes_by_phrase = {}
        scores_by_phrase = self._score_map(phrases, logits)
        for idx, phrase in enumerate(phrases):
            if idx < len(boxes_xyxy):
                boxes_by_phrase[phrase] = boxes_xyxy[idx]
        return boxes_by_phrase, scores_by_phrase

    def _merge_phrase_results(self, tracked, dino):
        tracked_boxes_by_phrase, tracked_scores_by_phrase = self._phrase_maps(tracked)
        dino_boxes_by_phrase, dino_scores_by_phrase = self._phrase_maps(dino)
        consistent = self._boxes_are_consistent(tracked[1], tracked[3], dino[1], dino[3])

        tracked_phrase_set = set(tracked_boxes_by_phrase)
        dino_phrase_set = set(dino_boxes_by_phrase)
        if len(dino_phrase_set) > len(tracked_phrase_set):
            logger.debug(
                "Rejecting DINO merge because it introduced extra phrases: tracked=%s dino=%s",
                sorted(tracked_phrase_set),
                sorted(dino_phrase_set),
            )
            return tracked, False

        merged_phrases = []
        merged_boxes = []
        merged_scores = []
        used_dino_any = False

        phrase_order = []
        for phrase in tracked[3]:
            if phrase not in phrase_order:
                phrase_order.append(phrase)
        for phrase in dino[3]:
            if phrase not in phrase_order:
                phrase_order.append(phrase)

        for phrase in phrase_order:
            has_tracked = phrase in tracked_boxes_by_phrase
            has_dino = phrase in dino_boxes_by_phrase

            if has_tracked and has_dino:
                old_score = tracked_scores_by_phrase.get(phrase, float("-inf"))
                new_score = dino_scores_by_phrase.get(phrase, float("-inf"))
                tracked_box = tracked_boxes_by_phrase[phrase]
                dino_box = dino_boxes_by_phrase[phrase]
                iou = self._single_box_iou(tracked_box, dino_box)
                center_dist = self._single_box_center_distance(tracked_box, dino_box)
                consistent_phrase = not (iou < self.dino_accept_iou and center_dist > self.dino_accept_center_dist)
                merged_score = max(old_score, new_score)

                if consistent_phrase:
                    merged_phrases.append(phrase)
                    merged_boxes.append(dino_box)
                    merged_scores.append(merged_score)
                    logger.debug(
                        "Phrase gate phrase='%s': consistent use DINO box, merged_score=max(%.4f, %.4f)=%.4f",
                        phrase,
                        old_score,
                        new_score,
                        merged_score,
                    )
                    if new_score > old_score:
                        used_dino_any = True
                    continue

                accept_dino = new_score > old_score + self.dino_score_margin
                logger.debug(
                    "Phrase gate phrase='%s': consistent=%s old=%.4f new=%.4f accept_dino=%s",
                    phrase,
                    consistent_phrase,
                    old_score,
                    new_score,
                    accept_dino,
                )
                if accept_dino:
                    merged_boxes.append(dino_box)
                    merged_scores.append(new_score)
                    used_dino_any = True
                else:
                    merged_boxes.append(tracked_box)
                    merged_scores.append(old_score)
                merged_phrases.append(phrase)
                continue

            if has_tracked:
                merged_phrases.append(phrase)
                merged_boxes.append(tracked_boxes_by_phrase[phrase])
                merged_scores.append(tracked_scores_by_phrase.get(phrase, 1.0))
                logger.debug("Phrase gate phrase='%s': tracker-only keep tracker", phrase)
                continue

            if has_dino:
                merged_phrases.append(phrase)
                merged_boxes.append(dino_boxes_by_phrase[phrase])
                merged_scores.append(dino_scores_by_phrase.get(phrase, 0.0))
                used_dino_any = True
                logger.debug("Phrase gate phrase='%s': dino-only accept dino", phrase)

        if not merged_boxes:
            return tracked, False

        boxes_xyxy = torch.stack(
            [box.detach().cpu() if hasattr(box, "detach") else torch.as_tensor(box, dtype=torch.float32) for box in merged_boxes]
        )
        boxes_cxcywh = self._xyxy_to_cxcywh(boxes_xyxy)
        logits = torch.tensor(merged_scores, dtype=torch.float32)
        merged_result = (
            boxes_cxcywh,
            boxes_xyxy,
            logits,
            merged_phrases,
            self._ensure_frame(tracked[4] if len(tracked) > 4 else None),
        )
        return merged_result, used_dino_any

    def get_dino_boxes(self, image, text_prompt):
        prompt_key = self._normalize_prompt(text_prompt)
        self.box_query_count += 1

        need_refresh = self._should_refresh_dino(prompt_key)

        tracked = None
        if self.cached_boxes is not None and self.tracker_prompt == prompt_key and self.trackers:
            try:
                tracked = self.get_tracked_boxes(image, text_prompt)
                if self._has_boxes(tracked) and not need_refresh:
                    self.last_query_used_dino = False
                    return tracked
                if not self._has_boxes(tracked):
                    tracked = None
            except Exception:
                tracked = None

        dino = self._raw_dino_candidate(image, text_prompt)

        if not self._has_boxes(dino):
            if tracked is not None and self._has_boxes(tracked):
                self.last_query_used_dino = False
                return tracked
            if self.cached_boxes is not None:
                self.last_query_used_dino = False
                return self.cached_boxes
            self.last_query_used_dino = True
            return dino

        if tracked is None or not self._has_boxes(tracked):
            return self._commit_result(image, dino, prompt_key, used_dino=True)

        tracked_boxes_xyxy = tracked[1]
        tracked_phrases = tracked[3]
        dino_boxes_xyxy = dino[1]
        dino_phrases = dino[3]

        merged_result, used_dino_any = self._merge_phrase_results(tracked, dino)
        return self._commit_result(image, merged_result, prompt_key, used_dino=used_dino_any)

class VLASHVectorAsyncManager:
    """
    针对向量化环境的异步管理器
    支持：多线程预取 (Hungry Strategy)、Batch推理、自动 Chunk 切换、自适应chunk size
    """
    def __init__(self, policy: PreTrainedPolicy, overlap_steps: int, num_envs: int, task_name, cfg: VLASHEvalConfig):
        self.policy = policy
        self.n_action_steps = policy.config.n_action_steps
        self.overlap_steps = overlap_steps
        self.num_envs = num_envs
        self.device = get_safe_torch_device(policy.config.device)
        self.task_name = task_name
        self.cfg = cfg
        
        # 初始化自适应策略
        if cfg.use_adaptive_chunk:
            env_continuity_threshold = cfg.adaptive_continuity_threshold
            # if cfg.env_task == "libero_object":
            #     env_continuity_threshold = 0.2
            # elif cfg.env_task == "libero_goal":
            #     env_continuity_threshold = 0.25
            self.adaptive_chunker = AdaptiveActionChunker(
                min_horizon=cfg.adaptive_min_horizon,
                max_horizon=cfg.adaptive_max_horizon,
                uncertainty_threshold=cfg.adaptive_uncertainty_threshold,
                coverage_threshold=cfg.adaptive_coverage_threshold,
                continuity_threshold=env_continuity_threshold,
            )
            # 初始chunk size
            self.current_chunk_size = self.adaptive_chunker.current_horizon
        else:
            self.adaptive_chunker = None
            self.current_chunk_size = self.n_action_steps
        self.valid_chunk_size = self.current_chunk_size
        self.last_chunk_size = self.current_chunk_size
        
        # 线程池：用于后台执行推理任务和自适应任务
        self.executor = concurrent.futures.ThreadPoolExecutor(max_workers=3)  # 增加线程用于后台处理
        self.next_chunk_future = None
        self.background_future = None
        
        # 状态追踪
        self.current_chunks = None  # 形状: [Batch, Horizon, ActionDim]
        self.chunk_index = 0
        
        # 校验
        assert self.n_action_steps > self.overlap_steps
        
        self.last_observation_imgs = None
        self.save_attention_step = 5
        
        self.video_images = []
        self.target_layers = [9, 10, 11, 12]
        # self.target_layers = [12]
        self.video_heatmap = []
        
        # 视频保存相关
        self.save_video = cfg.save_video
        self.video_output_dir = cfg.video_output_dir
        
        self.indexes = None
        self.inference_times = 0
        self.batch_env_step = 0
        self.last_launch_step = None
        self.launch_step_intervals = []

    def reset_state(self):
        """每个 Batch 开始前调用"""
        if self.next_chunk_future:
            self.next_chunk_future.cancel()
        if self.background_future:
            self.background_future.cancel()
        self.next_chunk_future = None
        self.background_future = None
        self.current_chunks = None
        self.chunk_index = 0
        # 重置视频帧列表
        self.video_images = []
        # 添加与 rollout.py 一致的 policy.reset() 调用
        self.policy.reset()
        if self.adaptive_chunker:
            self.adaptive_chunker.reset()
            self.current_chunk_size = self.adaptive_chunker.current_horizon
        else:
            self.current_chunk_size = self.n_action_steps
        self.inference_times = 0
        self.batch_env_step = 0
        self.last_launch_step = None

    def _record_launch_interval(self):
        if self.last_launch_step is not None:
            interval = self.batch_env_step - self.last_launch_step
            self.launch_step_intervals.append(interval)
        self.last_launch_step = self.batch_env_step

    def get_launch_interval_stats(self):
        if not self.launch_step_intervals:
            return {
                "count": 0,
                "intervals": [],
                "distribution": {},
                "mean": 0.0,
                "min": None,
                "max": None,
            }
        dist = Counter(self.launch_step_intervals)
        return {
            "count": len(self.launch_step_intervals),
            "intervals": list(self.launch_step_intervals),
            "distribution": dict(sorted(dist.items())),
            "mean": float(np.mean(self.launch_step_intervals)),
            "min": int(min(self.launch_step_intervals)),
            "max": int(max(self.launch_step_intervals)),
        }

    def save_video_frames(self, filename):
        """保存视频"""
        if not self.video_images:
            logging.info("No video frames to save.")
            return

        # 创建视频输出目录
        video_dir = Path(self.video_output_dir)
        video_dir.mkdir(parents=True, exist_ok=True)

        # 获取视频帧的尺寸
        height, width, _ = self.video_images[0].shape

        # 创建视频写入器
        video_path = video_dir / f"{filename}.mp4"
        fourcc = cv2.VideoWriter_fourcc(*'mp4v')
        out = cv2.VideoWriter(str(video_path), fourcc, self.cfg.fps, (width, height))

        # 写入视频帧
        for img in self.video_images:
            # 确保图像是BGR格式
            if img.shape[-1] == 3:
                # 如果是RGB格式，转换为BGR
                frame = cv2.cvtColor(img, cv2.COLOR_RGB2BGR)
            else:
                frame = img
            out.write(frame)

        # 释放视频写入器
        out.release()
        logging.info(f"Video saved to {video_path}")
        self.video_images = []

    def _prepare_batch_obs(self, raw_obs: Dict[str, Any], task_description: str):
        """
        将 VectorEnv 的原始观测转换为 Policy 需要的 Batch Tensor 格式
        """
        # 注意：VectorEnv 返回的 raw_obs 已经是 [B, ...] 格式了
        
        # 为了兼容 prepare_observation_for_inference，我们手动构造 Batch 数据
        batch_input = {}
        
        # 1. 处理状态 (Proprioception)
        if "agent_pos" in raw_obs:
            # raw_obs["agent_pos"] 形状是 [B, StateDim]
            batch_input["observation.state"] = torch.from_numpy(raw_obs["agent_pos"]).float().to(self.device)
        
        # 2. 处理图像
        if "pixels" in raw_obs and isinstance(raw_obs["pixels"], dict):
            for k, v in raw_obs["pixels"].items():
                '''
                image (1, 256, 256, 3)
                image2 (1, 256, 256, 3)
                '''
                # v 形状是 [B, H, W, C] -> 需要转换为 [B, C, H, W] 并归一化
                img_tensor = torch.from_numpy(v).float().permute(0, 3, 1, 2).to(self.device) / 255.0
                batch_input[f"observation.images.{k}"] = img_tensor
        
        # 3. 处理任务描述
        # 策略通常期望任务描述是一个 List[str] 或统一的字符串
        batch_input["task"] = [task_description] * self.num_envs
        
        return batch_input

    def _inference_task(self, batch_obs_prepared):
        """在后台线程执行的推理函数"""
        with torch.inference_mode():
            actions = self.policy.predict_action_chunk(batch_obs_prepared)
            return actions.cpu().numpy() # [B, Horizon, ActionDim]

    def _background_adaptive_task(self, image, instruction, robot_state=None):
        """在后台线程执行的自适应chunk size调整"""
        if self.adaptive_chunker:
            # 使用启发式策略调整chunk size
            _, info = self.adaptive_chunker.adaptive_chunk_heuristic(image, instruction, robot_state)
            # 注意：这里不直接更新current_chunk_size，而是在主线程中处理
            return info
        return {}

    def launch_next_inference(self, raw_obs: Dict[str, Any], task_description: str):
        """启动后台推理"""
        self.inference_times += 1
        self._record_launch_interval()
        if self.adaptive_chunker is not None:
            logging.info(f"Launching inference for task")
        self.save_attention_step -= 1
        prepared_obs = self._prepare_batch_obs(raw_obs, task_description)
        self.valid_chunk_size = self.current_chunk_size
        # 启动推理任务
        self.next_chunk_future = self.executor.submit(self._inference_task, prepared_obs)

    def reset_chunk(self, reason=False):
        self.current_chunks = self.next_chunk_future.result()
        self.last_chunk_size = self.valid_chunk_size
        self.next_chunk_future = None
        if self.adaptive_chunker is not None:
            self.adaptive_chunker.action_history = []
            if reason is False:
                self.get_gripper_change_index()
        
    def get_gripper_change_index(self):
        """获取当前 Chunk 中 gripper 动作变化的索引"""
        # 计算 gripper 动作的变化量
        gripper_changes = self.current_chunks[:, 1:, -1] - self.current_chunks[:, :-1, -1]
        # 找到第一个变化大于阈值的索引
        # 这个where返回的是元祖，第一个元素是满足条件的行索引，第二个元素是满足条件的列索引
        self.indexes = np.where(gripper_changes > 1.5)[1][0] if np.any(gripper_changes > 1.5) else None
        logging.info(f"Gripper change index: {self.indexes}")
        
    def should_interrupt_gripper(self):
        if self.adaptive_chunker and self.indexes is not None:
            if self.chunk_index + 5 == self.indexes:
            # if self.chunk_index + 5 <= self.indexes and self.chunk_index + 10 >= self.indexes: # 提前5步到后10步内预测到有夹爪变化
                logging.info(f"Interrupting current chunk due to predicted gripper change at index {self.indexes}")
                self.indexes = None
                return True
            else:
                return False
        return False

    def get_action_batch(self, raw_obs: Dict[str, Any], task_description: str) -> np.ndarray:
        """
        获取当前步的 Batch Action
        """
        if self.save_video:
            if 'image' in raw_obs["pixels"]:
                self.video_images.append(raw_obs["pixels"]['image'][0])
        should_launch = False
        gripper_change = False
        # 启动后台自适应任务（每次调用都会执行，比inference频率更高）
        if self.adaptive_chunker and "pixels" in raw_obs and isinstance(raw_obs["pixels"], dict):
            # 获取当前图像
            image = raw_obs["pixels"]['image'][0]  # 取第一个环境的图像
            
            # 构建机器人状态（如果可用）
            robot_state = None
            if "agent_pos" in raw_obs:
                # 取第一个环境的机器人位置, 我们 假设使用时batch大小为1
                agent_pos = raw_obs["agent_pos"][0]
                robot_state = {
                    "ee_position": agent_pos[:3],  # 假设前三个元素是末端执行器位置
                }
            
            self.background_future = self.executor.submit(self._background_adaptive_task, image, task_description, robot_state)
        
        # 第一次和后面的逻辑合一了
        if self.current_chunks is None:
            if self.next_chunk_future is None:
                self.launch_next_inference(raw_obs, task_description)
                self.chunk_index = 0
            else:
                self.chunk_index = self.overlap_steps
                
            # 阻塞等待结果（如果是预取的，这里会立即返回；如果是同步的，这里会等待）
            self.reset_chunk()
            # self.current_chunks = self.next_chunk_future.result()
            # self.next_chunk_future = None
            
            # 检查后台自适应任务是否完成
            if self.background_future:
                try:
                    background_info = self.background_future.result(timeout=1.0)  # 最多等待0.1秒
                    # 更新当前chunk size
                    self.current_chunk_size = self.adaptive_chunker.current_horizon
                    logging.info(f"Adaptive chunk size: {self.current_chunk_size}")
                except concurrent.futures.TimeoutError:
                    print(f"Adaptive chunk size background task timed out: {background_info}")
                    pass  # 超时不影响主流程
                self.background_future = None
            
            # 获取注意力图并使用它进一步调整chunk size (只有在policy真正做chunk推理时才使用ValueFunc)
            if self.adaptive_chunker:
                try:
                    attn_maps = self.policy.get_attn_maps(layers=self.target_layers, heads=range(8), force=True)
                    # 使用注意力图调整chunk size
                    if attn_maps:
                        vla_attention_map = attn_maps #[layers, num_heads, seq_len]
                        # 使用ValueFunc根据注意力图调整chunk size
                        _, info = self.adaptive_chunker.ValueFunc(
                            raw_obs["pixels"]['image'][0], 
                            task_description, 
                            vla_attention_map
                        )
                        # 更新当前chunk size
                        self.current_chunk_size = self.adaptive_chunker.current_horizon
                        # 立马根据Value func更新last_chunk_size，仅用当前指定的chunk size
                        self.last_chunk_size = min(self.current_chunk_size, self.last_chunk_size)
                        # 检查是否需要立即打断当前chunk
                        if hasattr(self.adaptive_chunker, 'needs_interruption') and self.adaptive_chunker.needs_interruption:
                            # 立即打断当前chunk，重置状态
                            logging.info(f"Interrupting current chunk due to attention-based adjustment")
                            should_launch = True
                            # self.current_chunks = None
                            # self.chunk_index = 0
                            # 立即启动新的推理
                            # self.launch_next_inference(raw_obs, task_description)
                        logging.info(f"Attention-based chunk size: {self.current_chunk_size}")
                except Exception as e:
                    import traceback
                    logging.warning(f"Failed to get attention maps: {e}")
                    traceback.print_exc()
            
            # self.video_heatmap.append(attn_maps)
            # if self.save_attention_step == 0 and self.last_observation_imgs is not None:
            #     # 可视化注意力
            #     see_layers = range(17)
            #     see_heads = range(8)
            #     attn_maps = self.policy.get_attn_maps(layers=range(17), heads=range(8))
            #     visualize_vla_attention_sum_heads_per_layer(
            #         images=self.last_observation_imgs,
            #         attn_maps=attn_maps,
            #         img_start_indices=[0, 256],
            #         output_dir='visualizations/8to14everylayer'+self.task_name
            #     )
            #     save_images(self.last_observation_imgs, filename=self.task_name)
        else:
            # 检查后台自适应任务是否完成
            if self.background_future:
                try:
                    background_info = self.background_future.result(timeout=1.0)  # 最多等待0.05秒
                    # 更新当前chunk size
                    self.current_chunk_size = self.adaptive_chunker.current_horizon
                    # 检查是否需要立即打断当前chunk
                    if hasattr(self.adaptive_chunker, 'needs_interruption') and self.adaptive_chunker.needs_interruption:
                        # 立即打断当前chunk，重置状态并启动新的预测
                        logging.info(f"Interrupting current chunk due to: {background_info.get('factors', 'N/A')}")
                        should_launch = True
                        # self.chunk_index = 0
                        # # 立即启动新的推理
                        # self.launch_next_inference(raw_obs, task_description)
                    # 每5步记录一次启发式调整
                    # if self.chunk_index % 5 == 0:
                    # logging.info(f"Heuristic-based chunk size: {self.current_chunk_size}, Reason: {background_info.get('factors', 'N/A')}")
                    if self.should_interrupt_gripper():
                        should_launch = True
                        gripper_change = True
                except concurrent.futures.TimeoutError:
                    pass  # 超时不影响主流程
                self.background_future = None

        # --- 3. 预取逻辑 (Hungry Strategy) ---
        # 当执行到当前 Chunk 的特定位置时，提前启动下一波推理
        if self.overlap_steps > 0:
            overlap_point = min(self.current_chunk_size, self.n_action_steps, self.last_chunk_size) - self.overlap_steps
            if self.chunk_index == overlap_point and self.next_chunk_future is None:
                logging.info(f"Manager: Reached overlap point ({overlap_point}), launching prefetch...")
                self.launch_next_inference(raw_obs, task_description)

        # --- 4. 提取当前步 Action ---
        # 打断开始推理
        if should_launch:
            if self.next_chunk_future is None:
                # 这个应该只需要current chunk size > overlapped就行 
                # 重置模型的KV缓存，避免注意力掩码大小不匹配
                # for layer in self.policy.model.layers:
                #     layer.self_attn.attn.reset_cache()
                logging.info(f"Manager: Interrupting current chunk with chunk size: {self.current_chunk_size}")    
                self.launch_next_inference(raw_obs, task_description)
                # 还要继续执行吗 执行的话就是index继续增加, 欸但是貌似这个没有起作用吧
                # 因为后续chunk size可能增加会来了，那还是打断吧
                
            # 阻塞等待结果 #这里如果打断了需要同步吗
            self.reset_chunk(gripper_change)
            # self.current_chunks = self.next_chunk_future.result() #md还是回出错在这里
            # self.next_chunk_future = None
            # # self.chunk_index = self.overlap_steps
            self.chunk_index = 0
        
        # 确保不超出当前chunk size
        current_step = min(self.chunk_index, min(self.current_chunk_size, self.n_action_steps, self.last_chunk_size) - 1)
        actions = self.current_chunks[:, current_step, :] # [B, ActionDim]
        # print("gripper action: ", actions[0, -1])

        if self.adaptive_chunker:
            self.adaptive_chunker.enqueue_history(actions[0].tolist())

        # logging.info(f"Manager: Current chunk index: {self.chunk_index}")

        # overlapped=4且min horizon=5时会导致一但进入5，就会持续每步预测，然后不连续导致进一步维持在5
        self.chunk_index = (self.chunk_index + 1)
        # logging.info(f"last chunk size: {self.last_chunk_size}, valid chunk size: {self.valid_chunk_size}")
        # 当达到当前chunk size或最大步骤时，重置
        if self.chunk_index >= min(self.current_chunk_size, self.n_action_steps, self.last_chunk_size):
            self.current_chunks = None
            self.chunk_index = 0
        self.batch_env_step += 1
        return actions

def eval_single_vec_env(
    vec_env: gym.vector.VectorEnv,
    policy: PreTrainedPolicy,
    cfg: VLASHEvalConfig,
    task_description: str,
    task_name: str
) -> Dict[str, Any]:
    
    effective_overlap = cfg.inference_overlap_steps * cfg.action_quant_ratio
    
    # 初始化异步 Batch 管理器
    async_mgr = VLASHVectorAsyncManager(
        policy=policy,
        overlap_steps=effective_overlap,
        num_envs=cfg.eval_batch_size,
        task_name=task_name,
        cfg=cfg
    )

    total_episodes = cfg.max_episodes
    n_batches = (total_episodes + cfg.eval_batch_size - 1) // cfg.eval_batch_size
    
    successes = []
    episode_steps = []
    max_steps = vec_env.call("_max_episode_steps")[0] #这里有max_steps
    monitor = GPUMonitor(gpu_id=0, interval=0.05)
    if cfg.use_monitor:
        monitor.start()
    for b in range(n_batches):
        # 使用全局种子 + 批次索引作为种子，确保每次运行结果一致
        batch_seed = cfg.seed + b
        observation, info = vec_env.reset(seed=batch_seed)
        async_mgr.reset_state()
        
        dones = np.array([False] * cfg.eval_batch_size)
        batch_success = np.array([False] * cfg.eval_batch_size)
        batch_steps = np.zeros(cfg.eval_batch_size, dtype=int)
        
        pbar = trange(max_steps, desc=f" Batch {b+1}/{n_batches}", leave=False, disable=inside_slurm())
        for step in pbar:
            if np.all(dones): break
            
            # 使用异步管理器直接获取 Batch Actions
            # 它内部会自动处理：
            # 1. 什么时候该在后台跑推理
            # 2. 什么时候该等待推理结果并切换 Chunk
            # 3. 后台运行Grounding DINO并调整chunk size
            # 4. 根据attention map进一步调整chunk size
            actions = async_mgr.get_action_batch(observation, task_description)
            
            observation, reward, terminated, truncated, info = vec_env.step(actions)
            
            # 增加步骤计数
            batch_steps[~dones] += 1
            
            # 成功率统计逻辑
            if "final_info" in info:
                final_info = info["final_info"]
                if isinstance(final_info, dict) and "is_success" in final_info:
                    is_success_array = final_info["is_success"]
                    just_finished = (terminated | truncated) & (~dones)
                    if np.any(just_finished):
                        batch_success = np.where(just_finished, is_success_array, batch_success)
            
            dones = terminated | truncated | dones

        successes.extend(batch_success.tolist())
        episode_steps.extend(batch_steps.tolist())

        # 保存视频
        if cfg.save_video:
            async_mgr.save_video_frames(f"{task_name}_env0_batch{b}")
    if cfg.use_monitor:
        monitor.stop()
        stats = monitor.get_report() 
        logging.info(f"inference times: {async_mgr.inference_times}")
        launch_interval_stats = async_mgr.get_launch_interval_stats()
        logging.info(f"Launch inference interval stats: {launch_interval_stats}")
        logging.info(f"GPU Monitor Stats: {stats}")
    else:
        launch_interval_stats = async_mgr.get_launch_interval_stats()
        logging.info(f"inference times: {async_mgr.inference_times}")
        logging.info(f"Launch inference interval stats: {launch_interval_stats}")
    
    successes = successes[:total_episodes]
    
    # from visualutils import heatmap_videos
    # heatmap_videos(async_mgr.video_images, async_mgr.video_heatmap, filename='visualizations/'+task_name+".mp4")
    episode_steps = episode_steps[:total_episodes]
    return {
        "sr": np.mean(successes),
        "count": len(successes),
        "success_list": successes,
        "steps_list": episode_steps,
        "avg_steps": np.mean(episode_steps) if episode_steps else 0,
        "inference_times": async_mgr.inference_times,
        "launch_interval_stats": launch_interval_stats,
    }
    
@parser.wrap()
def run_benchmark(cfg: VLASHEvalConfig):
    init_logging()
    _configure_component_log_levels(cfg)
    set_seed(cfg.seed)
    
    # 添加与 rollout.py 相同的 torch 后端设置，确保结果一致
    torch.backends.cudnn.benchmark = True
    torch.backends.cuda.matmul.allow_tf32 = True
    
    logging.info(f"max_episodes: {cfg.max_episodes}, inference_overlap_steps: {cfg.inference_overlap_steps}, ")
    logging.info(f"Adaptive chunk size enabled: {cfg.use_adaptive_chunk}")
    if cfg.use_adaptive_chunk:
        logging.info(f"Adaptive chunk size range: {cfg.adaptive_min_horizon} - {cfg.adaptive_max_horizon}")
    
    from vlash.train import make_vlash_dataset
    dataset = make_vlash_dataset(cfg.train_cfg)
    
    # 1. 加载策略
    policy_cls = get_policy_class(cfg.policy.type)
    # print(dataset.meta.stats)
    policy = policy_cls.from_pretrained(cfg.policy.pretrained_path, config=cfg.policy, dataset_stats=dataset.meta.stats) 
    policy.eval()
    # 2. 初始化所有任务的环境
    if cfg.env_type == "libero":
        # 不指定 task_ids，让 factory 自动创建该 suite 下的所有任务
        env_cfg = LiberoEnv(task=cfg.env_task, task_ids=cfg.task_ids) 
    else:
        raise ValueError(f"Env type {cfg.env_type} not supported")

    logging.info(f"Initializing all tasks for {cfg.env_task}...")
    # make_env 返回 {task_group: {task_id: VectorEnv}}
    all_envs_dict = make_env(env_cfg, n_envs=cfg.eval_batch_size)

    # 3. 遍历评估
    results_per_task = {}
    overall_successes = []
    start_time = time.time()
    task_name = ""

    for suite_name, tasks in all_envs_dict.items():
        logging.info(f"Evaluating Suite: {suite_name}")
        for task_id, vec_env in tasks.items():
            # 从环境元数据中提取任务指令 (Libero 特有)
            # 如果配置中指定了 single_task 则覆盖，否则从环境取
            try:
                task_description = vec_env.get_attr("task_description")[0]
            except Exception:
                task_description = cfg.single_task
                
            logging.info(f"Starting Task [{task_id}]: {task_description}")
            
            task_name = suite_name + str(task_id)
            
            task_info = eval_single_vec_env(vec_env, policy, cfg, task_description, task_name=task_name)
            
            results_per_task[f"{suite_name}_{task_id}"] = {
                "task_description": task_description,
                "success_rate": task_info["sr"],
                "num_episodes": task_info["count"],
                "avg_steps": task_info["avg_steps"],
                "steps_list": task_info["steps_list"],
                "inference_times": task_info["inference_times"],
                "launch_interval_stats": task_info["launch_interval_stats"],
            }
            overall_successes.extend(task_info["success_list"])
            
            logging.info(f"Task Done. SR: {task_info['sr']*100:.1f}%")

    # 4. 汇总与保存
    total_time = time.time() - start_time
    
    # 收集所有episode的步骤
    all_steps = []
    for task_result in results_per_task.values():
        all_steps.extend(task_result["steps_list"])
    
    final_results = {
        "overall": {
            "avg_success_rate": np.mean(overall_successes),
            "total_episodes": len(overall_successes),
            "total_eval_time_s": total_time,
            "env_task": cfg.env_task,
            "use_adaptive_chunk": cfg.use_adaptive_chunk,
            "avg_steps_per_episode": np.mean(all_steps) if all_steps else 0,
            "total_steps": sum(all_steps) if all_steps else 0
        },
        "per_task": results_per_task
    }
    
    out_dir = Path(cfg.output_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    with open(out_dir / "full_benchmark_results.json", "w") as f:
        json.dump(final_results, f, indent=2)
    
    logging.info("="*50)
    logging.info(f"FINAL OVERALL SR: {final_results['overall']['avg_success_rate']*100:.2f}%")
    logging.info(f"Results saved to {out_dir}")
    logging.info("="*50)

    close_envs(all_envs_dict)
    
    num_tasks = len(results_per_task)
    # 每个任务测多少
    ep_per_task = cfg.max_episodes
    # 总测试数
    total_test_count = len(overall_successes)

    logging.info("="*50)
